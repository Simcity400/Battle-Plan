on_actions = {
	on_startup = {
		effect = {
			random_country = {
				# I want those variable to initalize globally on launch
				#####################################################################
				#2.0 release 
				#####################################################################
				# --- Battle Plan: bind the 8 graph curves to the HUMAN players ---
				# (was hardcoded GER/ITA/USA/SOV/FRA/ENG/JAP/CHI; those tags are not
				#  all present in AMSB, so half the curves used to bind to dead tags.)
				#
				# Every country always accumulates its own ASG_*_score_array, so a
				# colour slot is just a pointer: writing a player's .id here makes
				# that curve replot their full history. We store .id (a numeric
				# handle), never a country-object var, so a player capitulating can
				# never cause an annex-then-deref.
				#
				# Unfilled slots (fewer than 8 humans) default to WST (the always-
				# alive wasteland tag) = a harmless flat-zero line. The Phase 2 host
				# picker can repoint any slot at any country live.
				set_variable = { GLOBAL.ASG_green_curve_tag = WST.id }
				set_variable = { GLOBAL.ASG_blue_curve_tag = WST.id }
				set_variable = { GLOBAL.ASG_grey_curve_tag = WST.id }
				set_variable = { GLOBAL.ASG_red_curve_tag = WST.id }
				set_variable = { GLOBAL.ASG_light_blue_curve_tag = WST.id }
				set_variable = { GLOBAL.ASG_light_red_curve_tag = WST.id }
				set_variable = { GLOBAL.ASG_light_white_curve_tag = WST.id }
				set_variable = { GLOBAL.ASG_light_yellow_curve_tag = WST.id }
				# Fill the colour slots with the first 8 humans found (deterministic
				# every_country order). Swap the slot order here to taste.
				set_variable = { GLOBAL.ASG_bind_index = 0 }
				every_country = {
					limit = { is_ai = no }
					if = {
						limit = { check_variable = { GLOBAL.ASG_bind_index = 0 } }
						set_variable = { GLOBAL.ASG_green_curve_tag = THIS.id }
					}
					else_if = {
						limit = { check_variable = { GLOBAL.ASG_bind_index = 1 } }
						set_variable = { GLOBAL.ASG_blue_curve_tag = THIS.id }
					}
					else_if = {
						limit = { check_variable = { GLOBAL.ASG_bind_index = 2 } }
						set_variable = { GLOBAL.ASG_grey_curve_tag = THIS.id }
					}
					else_if = {
						limit = { check_variable = { GLOBAL.ASG_bind_index = 3 } }
						set_variable = { GLOBAL.ASG_red_curve_tag = THIS.id }
					}
					else_if = {
						limit = { check_variable = { GLOBAL.ASG_bind_index = 4 } }
						set_variable = { GLOBAL.ASG_light_blue_curve_tag = THIS.id }
					}
					else_if = {
						limit = { check_variable = { GLOBAL.ASG_bind_index = 5 } }
						set_variable = { GLOBAL.ASG_light_red_curve_tag = THIS.id }
					}
					else_if = {
						limit = { check_variable = { GLOBAL.ASG_bind_index = 6 } }
						set_variable = { GLOBAL.ASG_light_white_curve_tag = THIS.id }
					}
					else_if = {
						limit = { check_variable = { GLOBAL.ASG_bind_index = 7 } }
						set_variable = { GLOBAL.ASG_light_yellow_curve_tag = THIS.id }
					}
					add_to_variable = { GLOBAL.ASG_bind_index = 1 }
				}
				set_variable = {
					GLOBAL.ASG_number_of_executed_ticks = 0
				}
				set_variable = {
					GLOBAL.ASG_civilian_rendering_scale_factor = 1
				}
				set_variable = {
					GLOBAL.ASG_military_rendering_scale_factor = 1
				}
				set_variable = {
					GLOBAL.ASG_manpower_rendering_scale_factor = 1
				}
				set_variable = {
					GLOBAL.ASG_casualties_rendering_scale_factor = 1
				}
				set_variable = {
					GLOBAL.ASG_graph_pixel_y_maximum_value = 390
				}
				set_variable = {
					GLOBAL.ASG_graph_pixel_y_minimum_value = 0
				}
				set_variable = {
					GLOBAL.ASG_graph_pixel_y_offset_value = 4
				}
				set_variable = {
					GLOBAL.ASG_manpower_amount_to_pixel_scale_factor = 0.04
				}
				set_variable = {
					GLOBAL.ASG_military_amount_to_pixel_scale_factor = 0.60
				}
				set_variable = {
					GLOBAL.ASG_civilian_amount_to_pixel_scale_factor = 0.60
				}
				set_variable = {
					GLOBAL.ASG_casualties_amount_to_pixel_scale_factor = 0.030
				}

				set_variable = {
					GLOBAL.ASG_graph_pixel_y_maximum_value_corrected_by_offset = GLOBAL.ASG_graph_pixel_y_maximum_value
				}
				add_to_variable = {
					GLOBAL.ASG_graph_pixel_y_maximum_value_corrected_by_offset = ASG_graph_pixel_y_offset_value
				}
				set_variable = {
					GLOBAL.ASG_graph_pixel_y_minimum_value_corrected_by_offset = GLOBAL.ASG_graph_pixel_y_minimum_value
				}
				add_to_variable = {
					GLOBAL.ASG_graph_pixel_y_minimum_value_corrected_by_offset = ASG_graph_pixel_y_offset_value
				}
				ASG_stats_update = yes
				#military
				ASG_calculate_initial_graph_point_military = yes
				#civilian
				ASG_calculate_initial_graph_point_civilian = yes
				#manpower
				ASG_calculate_initial_graph_point_manpower = yes
				#casualties
				ASG_calculate_initial_graph_point_casualties = yes
				# web-dashboard export: header + the tick-0 baseline snapshot
				ASG_export_to_log = yes
			}
		}
	}
	# Tick off WST (the always-alive wasteland tag) instead of GER: a plain
	# on_monthly fires once PER COUNTRY (would replot ~60x/month), and on_monthly_GER
	# silently dies the moment Germany is annexed. WST fires exactly once a month and
	# never disappears. Root is irrelevant - everything below uses global vars /
	# every_country. (Matches the AMSB WST-executor pattern used by gazala_on_actions.)
	on_monthly_WST = {
		effect = {
			# log = "Graph update THIS = [?THIS.GetName] ROOT = [?ROOT.GetName]"
			ASG_graphs_replot = yes
			ASG_stats_update = yes
			add_to_variable = {
				GLOBAL.ASG_number_of_executed_ticks = 1
			}
			# web-dashboard export (one HOI4MPSTAT line per country, this month)
			ASG_export_to_log = yes
			# if = {
			# 	limit = {
			# 		is_debug = yes
			# 	}
			# 	log = "ASG Update Stats"
			# }
		}
	}
}
